#include <iostream>
#include <cmath>
using namespace std;

// ===== ФУНКЦИЯ =====
double f(double x) {
    return x * x + 2;
}

int main() {
    setlocale(LC_ALL, "rus");
    double x0 = 0;      // начальная точка
    double t = 1;      // начальный шаг

    double f0 = f(x0);
    double fl = f(x0 - t);
    double fr = f(x0 + t);

    // Проверка условия (а): минимум внутри [x0-t, x0+t]
    if (fl >= f0 && f0 <= fr) {
        cout << "Интервал: [" << x0 - t << ", " << x0 + t << "]\n";
        return 0;
    }

    // Направление движения
    double d = t;
    if (fl < f0) d = -t;

    // Итерации
    double xp = x0, fp = f0;
    double xc = x0 + d, fc = f(xc);

    while (fc < fp) {
        d *= 2;
        double xn = xc + d;
        double fn = f(xn);
        if (fn >= fc) {
            cout << "Интервал: [" << min(xp, xn) << ", " << max(xp, xn) << "]\n";
            return 0;
        }
        xp = xc; fp = fc;
        xc = xn; fc = fn;
    }

    cout << "Интервал: [" << min(x0, xc) << ", " << max(x0, xc) << "]\n";
   
}
